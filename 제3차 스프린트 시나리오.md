   ```bash
   from flask import Flask, render_template, request
   from flask_cors import CORS
   from flask_sock import Sock
   import ssl
   import io
   import numpy as np
   import torch
   import whisper
   import soundfile as sf
   import json

   app = Flask(__name__,
       template_folder='./www',
       static_folder='./www',
       static_url_path='/'
   )
   CORS(app)  # 모든 도메인에서의 접근을 허용
   sock = Sock(app)
   
   # 여러 언어에 대한 Whisper 모델 로드 (메모리 효율성을 위해 'tiny' 사용)
   model = whisper.load_model("tiny")

   @app.route('/')
   def index():
       return render_template('index.html')

   @sock.route('/audio')
   def handle_audio(ws):
       while True:
           try:
               data = ws.receive()
               if data is None:
                   break
               
               # JSON 형식인지 확인 (언어 설정 등의 메타데이터)
               try:
                   json_data = json.loads(data)
                   if 'language' in json_data:
                       language = json_data['language']
                       ws.send(json.dumps({'status': 'language_set', 'language': language}))
                       continue
               except:
                   # JSON이 아닌 경우, 오디오 데이터로 처리
                   pass
               
               audio_stream = io.BytesIO(data)
               audio_stream.seek(0)  # 스트림의 시작으로 이동

               # 오디오 데이터를 .wav 파일로 저장
               with open('received_audio.wav', 'wb') as f:
                   f.write(audio_stream.read())

               # Whisper 모델에 .wav 파일을 전달하여 인식
               # 기본 언어를 한국어(ko)로 설정하거나 클라이언트에서 전달받은 언어 사용
               language = getattr(ws, 'language', 'ko')
               result = model.transcribe('received_audio.wav', language=language)
               
               # 결과 전송
               ws.send(json.dumps({
                   'status': 'transcription',
                   'text': result['text'],
                   'language': language
               }))
               
           except Exception as e:
               print(f'Error: {e}')
               ws.send(json.dumps({
                   'status': 'error',
                   'message': str(e)
               }))

   if __name__ == "__main__":
       app.run(host="0.0.0.0", port=3000, debug=True)
   EOL
   ```

   ```bash
   const ErrorHandler = ({ 
     error, 
     onRetry, 
     autoHideDuration = 6000,
     severity = 'error'

     useEffect(() => {

         onClose={handleClose}
   ```

3. **WebSocket 연결 관리 개선**:
   ```bash
   cat > frontend/src/components/WebSocketManager.js << 'EOL'
   import { useState, useRef, useEffect, useCallback } from 'react';

   const WebSocketManager = (initialUrl) => {
     const [isConnected, setIsConnected] = useState(false);
     const [isReconnecting, setIsReconnecting] = useState(false);
     const [error, setError] = useState(null);
     const [url, setUrl] = useState(initialUrl);
     const [reconnectAttempts, setReconnectAttempts] = useState(0);
     const MAX_RECONNECT_ATTEMPTS = 5;
     const RECONNECT_INTERVAL = 3000; // 3초
     
     const socketRef = useRef(null);
     const reconnectTimeoutRef = useRef(null);
     
     const connect = useCallback(() => {
       // 기존 연결 정리
       if (socketRef.current) {
         socketRef.current.close();
       }
       
       // 재연결 타이머 정리
       if (reconnectTimeoutRef.current) {
         clearTimeout(reconnectTimeoutRef.current);
         reconnectTimeoutRef.current = null;
       }
       
       try {
         setError(null);
         socketRef.current = new WebSocket(url);
         
         socketRef.current.onopen = () => {
           console.log('WebSocket 연결 성공');
           setIsConnected(true);
           setIsReconnecting(false);
           setReconnectAttempts(0);
         };
         
         socketRef.current.onclose = (event) => {
           console.log('WebSocket 연결 종료', event);
           setIsConnected(false);
           
           // 정상 종료가 아닐 경우 재연결 시도
           if (!event.wasClean && reconnectAttempts < MAX_RECONNECT_ATTEMPTS) {
             setIsReconnecting(true);
             setReconnectAttempts(prev => prev + 1);
             setError(`연결이 종료되었습니다. 재연결 시도 중... (${reconnectAttempts + 1}/${MAX_RECONNECT_ATTEMPTS})`);
             
             reconnectTimeoutRef.current = setTimeout(() => {
               connect();
             }, RECONNECT_INTERVAL);
           } else if (reconnectAttempts >= MAX_RECONNECT_ATTEMPTS) {
             setIsReconnecting(false);
             setError('최대 재연결 시도 횟수에 도달했습니다. 수동으로 재연결해 주세요.');
           }
         };
         
         socketRef.current.onerror = (error) => {
           console.error('WebSocket 오류:', error);
           setError('WebSocket 연결 중 오류가 발생했습니다.');
         };
       } catch (error) {
         console.error('WebSocket 연결 생성 실패:', error);
         setError('WebSocket 연결을 설정할 수 없습니다.');
         setIsConnected(false);
       }
     }, [url, reconnectAttempts]);
     
     // 초기 연결 및 URL 변경 시 재연결
     useEffect(() => {
       connect();
       
       // 컴포넌트 언마운트 시 정리
       return () => {
         if (socketRef.current) {
           socketRef.current.close();
         }
         if (reconnectTimeoutRef.current) {
           clearTimeout(reconnectTimeoutRef.current);
         }
       };
     }, [url, connect]);
     
     // 메시지 전송 함수
     const sendMessage = useCallback((data) => {
       if (socketRef.current && socketRef.current.readyState === WebSocket.OPEN) {
         try {
           socketRef.current.send(data);
           return true;
         } catch (error) {
           console.error('메시지 전송 중 오류:', error);
           setError('메시지를 전송할 수 없습니다.');
           return false;
         }
       } else {
         setError('WebSocket이 연결되어 있지 않습니다.');
         return false;
       }
     }, []);
     
     // JSON 데이터 전송 함수
     const sendJsonMessage = useCallback((data) => {
       try {
         const jsonString = JSON.stringify(data);
         return sendMessage(jsonString);
       } catch (error) {
         console.error('JSON 변환 중 오류:', error);
         setError('데이터를 JSON으로 변환할 수 없습니다.');
         return false;
       }
     }, [sendMessage]);
     
     // 메시지 수신 콜백 등록
     const onMessage = useCallback((callback) => {
       if (socketRef.current) {
         socketRef.current.onmessage = (event) => {
           try {
             // JSON 형식 확인 시도
             let data = event.data;
             try {
               data = JSON.parse(event.data);
             } catch {
               // JSON이 아닌 경우 원본 데이터 사용
             }
             callback(data);
           } catch (error) {
             console.error('메시지 처리 중 오류:', error);
             setError('메시지를 처리할 수 없습니다.');
           }
         };
       }
     }, []);
     
     return {
       isConnected,
       isReconnecting,
       error,
       setUrl,
       connect,
       sendMessage,
       sendJsonMessage,
       onMessage
     };
   };

   export default WebSocketManager;
   EOL
   ```

   ```bash
       });
     }, [onMessage, language]);
     
     // 웹소켓 URL 변경 처리
     const handleWebSocketUrlChange = (event) => {
       setWebsocketUrl(event.target.value);
     };
     
     // URL 변경 후 연결 업데이트
     const handleUrlUpdate = () => {
       setUrl(websocketUrl);
     };
     
     // 언어 변경 처리
     const handleLanguageChange = (event) => {
       const newLanguage = event.target.value;
       setLanguage(newLanguage);
       if (isConnected) {
         sendJsonMessage({ language: newLanguage });
       }
     };
     
     // 설정 패널 토글
     const toggleSettings = () => {
       setShowSettings(!showSettings);
     };
     
     // 녹음 완료 처리
     const handleRecordingComplete = useCallback((audioBlob) => {
       if (isConnected) {
         setProcessing(true);
         // 언어 설정 전송
         sendJsonMessage({ language });
         
         // 오디오 데이터 전송
         const reader = new FileReader();
         reader.onloadend = () => {
           const audioArrayBuffer = reader.result;
           sendMessage(audioArrayBuffer);
         };
         reader.readAsArrayBuffer(audioBlob);
       }
     }, [isConnected, sendJsonMessage, sendMessage, language]);
     
     // 탭 변경 처리
     const handleTabChange = (event, newValue) => {
       setCurrentTab(newValue);
     };
     
     // 모든 기록 삭제
     const handleClearTranscriptions = () => {
       setTranscriptions([]);
     };
     
     return (
       <div className="App">
         <AppBar position="static">
           <Toolbar>
             <Typography variant="h6" component="div" sx={{ flexGrow: 1 }}>
               음성 인식 애플리케이션
             </Typography>
             <IconButton 
               color="inherit" 
               onClick={toggleSettings}
               aria-label="설정"
             >
               <SettingsIcon />
             </IconButton>
           </Toolbar>
         </AppBar>
         
         <Container maxWidth="md">
           <Box mt={3}>
             {showSettings && (
               <Paper elevation={2} sx={{ p: 2, mb: 2 }}>
                 <Typography variant="h6" gutterBottom>
                   연결 설정
                 </Typography>
                 <Box display="flex" alignItems="center" mb={2}>
                   <Box flexGrow={1} mr={2}>
                     <TextField
                       fullWidth
                       label="WebSocket URL"
                       variant="outlined"
                       size="small"
                       value={websocketUrl}
                       onChange={handleWebSocketUrlChange}
                     />
                   </Box>
                   <Button 
                     variant="contained" 
                     onClick={handleUrlUpdate}
                     startIcon={<RefreshIcon />}
                   >
                     연결
                   </Button>
                 </Box>
                 
                 <Box display="flex" alignItems="center">
                   <FormControl variant="outlined" size="small" fullWidth>
                     <InputLabel>인식 언어</InputLabel>
                     <Select
                       value={language}
                       onChange={handleLanguageChange}
                       label="인식 언어"
                     >
                       <MenuItem value="ko">한국어</MenuItem>
                       <MenuItem value="en">영어</MenuItem>
                       <MenuItem value="ja">일본어</MenuItem>
                       <MenuItem value="zh">중국어</MenuItem>
                     </Select>
                   </FormControl>
                   <Box ml={2} display="flex" alignItems="center">
                     <Typography variant="body2" mr={1}>
                       상태:
                     </Typography>
                     <Box
                       component="span"
                       sx={{
                         width: 12,
                         height: 12,
                         borderRadius: '50%',
                         backgroundColor: isConnected ? 'success.main' : 'error.main',
                         display: 'inline-block'
                       }}
                     />
                   </Box>
                 </Box>
               </Paper>
             )}
             
             <Paper elevation={3} sx={{ p: 3, borderRadius: 2 }}>
               <Tabs 
                 value={currentTab} 
                 onChange={handleTabChange} 
                 centered
                 sx={{ mb: 2 }}
               >
                 <Tab label="음성 녹음" />
                 <Tab label="텍스트 기록" />
               </Tabs>
               
               <Divider sx={{ mb: 3 }} />
               
               {currentTab === 0 && (
                 <>
                   <Typography variant="h5" gutterBottom align="center">
                     {language === 'ko' ? '마이크로 말하면 텍스트로 변환됩니다' : 
                      language === 'en' ? 'Speak into the microphone to convert to text' :
                      language === 'ja' ? 'マイクに向かって話すとテキストに変換されます' :
                      '对着麦克风说话，将转换为文字'}
                   </Typography>
                   
                   <Box display="flex" justifyContent="center" my={3}>
                     <AudioRecorder 
                       onRecordingComplete={handleRecordingComplete} 
                       disabled={!isConnected || isReconnecting || processing}
                     />
                   </Box>
                   
                   {processing && (
                     <Box sx={{ width: '100%', mt: 2, mb: 2 }}>
                       <Typography variant="body2" align="center" gutterBottom>
                         음성을 텍스트로 변환 중...
                       </Typography>
                       <LinearProgress 
                         variant="determinate" 
                         value={processingProgress} 
                         sx={{ height: 8, borderRadius: 4 }}
                       />
                     </Box>
                   )}
                 </>
               )}
               
               {currentTab === 1 && (
                 <>
                   <Box display="flex" justifyContent="space-between" alignItems="center" mb={2}>
                     <Typography variant="h6">
                       변환된 텍스트 기록
                     </Typography>
                     <Button 
                       variant="outlined" 
                       color="error" 
                       size="small"
                       startIcon={<DeleteIcon />}
                       onClick={handleClearTranscriptions}
                       disabled={transcriptions.length === 0}
                     >
                       모두 지우기
                     </Button>
                   </Box>
                   
                   <Box sx={{ maxHeight: 400, overflow: 'auto', mt: 2 }}>
                     {transcriptions.length > 0 ? (
                       transcriptions.map((item, index) => (
                         <Paper 
                           key={index} 
                           elevation={1} 
                           sx={{ p: 2, mb: 1, backgroundColor: '#f9f9f9' }}
                         >
                           <Typography variant="body1">
                             {item.text}
                           </Typography>
                           <Typography variant="caption" color="text.secondary">
                             {new Date(item.timestamp).toLocaleTimeString()} | {
                               item.language === 'ko' ? '한국어' : 
                               item.language === 'en' ? '영어' :
                               item.language === 'ja' ? '일본어' :
                               item.language === 'zh' ? '중국어' : 
                               item.language
                             }
                           </Typography>
                         </Paper>
                       ))
                     ) : (
                       <Typography variant="body1" color="text.secondary" align="center">
                         아직 변환된 텍스트가 없습니다. 음성 녹음 탭에서 녹음을 시작하세요.
                       </Typography>
                     )}
                   </Box>
                 </>
               )}
             </Paper>
           </Box>
         </Container>
         
         <ErrorHandler 
           error={error} 
           onRetry={connect}
           severity={isReconnecting ? "warning" : "error"} 
         />
       </div>
     );
   }

   export default App;
   EOL
   ```

### 2. 충돌 확인

개발자 A의 PR이 먼저 병합된 후, 개발자 B가 PR을 생성하면 충돌이 발생합니다.

GitHub에서 "This branch has conflicts that must be resolved" 메시지가 표시됩니다.

### 3. 충돌 해결 (개발자 B)

```bash
# develop 브랜치 최신화
git checkout develop
git pull origin develop

# 작업 브랜치로 전환
git checkout feature/audio-visualization

# develop 브랜치 변경사항 병합
git merge develop

# 충돌 발생 메시지 확인
# 충돌 파일 편집 (VS Code 등의 에디터에서 충돌 마커 해결)

# 충돌 해결 후 커밋
git add .
git commit -m "Merge develop and resolve conflicts"
git push origin feature/audio-visualization
```

### 4. VS Code에서 충돌 해결하기

VS Code는 충돌을 시각적으로 해결할 수 있는 도구를 제공합니다.

1. VS Code에서 충돌이 있는 파일을 열면 다음과 같은 마커가 표시됩니다:
   ```
   <<<<<<< HEAD (현재 브랜치의 변경사항)
   // 개발자 B의 코드
   =======
   // 개발자 A의 코드 (develop 브랜치에 병합된)
   >>>>>>> develop
   ```

2. VS Code 인터페이스에서 제공하는 "Accept Current Change", "Accept Incoming Change", "Accept Both Changes", "Compare Changes" 옵션 중 하나를 선택합니다.

3. 필요한 경우 수동으로 코드를 편집하여 두 변경사항을 적절히 통합합니다.

4. 모든 충돌 마커(<<<<<<, =======, >>>>>>>)가 제거되었는지 확인합니다.

5. 파일을 저장하고 위의 Git 명령어를 사용하여 충돌 해결을 완료합니다.

이 실습을 통해 실제 협업 과정에서 발생할 수 있는 코드 충돌 상황을 경험하고 해결하는 방법을 배울 수 있습니다.
